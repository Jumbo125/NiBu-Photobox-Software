// Countdown
PB.captureUI.show("Capture_countdown", {
  text: "Nächstes Foto…",
  counter: 3
});

// Trigger
PB.captureUI.show("Capture_working_trigger", {
  text: "Die Kamera macht das Foto… bitte nicht bewegen."
});

// Working Capture
PB.captureUI.show("Capture_working_capture", {
  text: "Aufnahme läuft…"
});

// Working Render
PB.captureUI.show("Capture_working_render", {
  text: "Verarbeitung läuft…"
});

// Abort
PB.captureUI.show("Capture_working_abort", {
  text: "Abbruch läuft…"
});

// Allgemeiner Fehler
PB.captureUI.show("Capture_error", {
  text: "Allgemeiner Fehler",
  onClose: () => PB.captureUI.hideAll()
});

// Papierfehler
PB.captureUI.show("Capture_error_paper", {
  text: "Papier leer oder nicht ausreichend.",
  onClose: () => PB.captureUI.hideAll()
});

// Papierfehler mit HTML
PB.captureUI.show("Capture_error_paper", {
  html: "<strong>Remaining prints:</strong> 2<br><strong>Required:</strong> 4",
  onClose: () => PB.captureUI.hideAll()
});

// Druckerfehler
PB.captureUI.show("Capture_error_printer", {
  text: "Druckerfehler oder Drucker offline.",
  onClose: () => PB.captureUI.hideAll()
});

// Finish ohne Bild
PB.captureUI.show("Capture_finish", {
  text: "Fertig!"
});

// Finish mit Bild
PB.captureUI.show("Capture_finish_with_img", {
  text: "Fertig!",
  imgSrc: "./src/img/trigger_wait.svg",
  onClose: () => PB.captureUI.hideAll()
});

// Preview zwischen Shots
PB.captureUI.show("Capture_preview_between_shots", {
  imgSrc: "./src/img/trigger_wait.svg"
});
